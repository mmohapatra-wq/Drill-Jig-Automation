! ============================================================================
! Datum CSYS from a round edge + a plane  (run this INSIDE Creo, not PowerShell)
! ----------------------------------------------------------------------------
! Result: a coordinate system whose ORIGIN is the round hole-edge center, +Y is
! that edge's axis (aligned to original Y), and +X is projected from a picked plane.
!
! This is the operator's own recorded mapkey with ONE line added at the end
! (~ Activate stdbtn_1 = click OK) so it creates the csys and closes automatically.
!
! WHY THIS INSTEAD OF csysinator.cmd: a Creo mapkey with @PAUSE_FOR_SCREEN_PICK is
! played BY CREO, which natively pauses for each mouse pick and resumes. Driving the
! same dialog piecemeal from PowerShell (RunMacro + COM reads) is what froze - a COM
! read while the dialog is open deadlocks. Running the mapkey in Creo avoids all of it.
!
! HOW TO USE:
!   1. Copy the "mapkey" block below into your config.pro
!      (Creo: File > Options > Configuration Editor > add, or edit config.pro directly).
!      OR paste it via Tools > Mapkeys... > New/Edit.
!   2. Load it: File > Options > ... > Import/Load config.pro, or restart Creo.
!   3. Open the PART. Run the mapkey named "datum" (type its trigger, or add it to a
!      toolbar/keyboard shortcut via Tools > Mapkeys).
!   4. It stops at pick 1 -> click the ROUND hole edge (origin). It stops at pick 2 ->
!      click the PLANE for +X (e.g. TOP). Then it flips the axes and clicks OK.
!
! If an axis comes out reversed on a differently-drilled hole, the flip line for that
! axis (~ Activate ... t2.FlipBtn1  for Y,  t2.FlipBtn2  for X) is what to add/remove.
! To inspect before it commits, delete the final "~ Activate `Odui_Dlg_00` `stdbtn_1`;"
! line and click OK yourself.
! ============================================================================
mapkey datum @MAPKEY_LABELdatum;~ Command `ProCmdDatumCsys` ;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t1.OriginPlacement` 2 `DuMmY` `constr`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t1.OriginPlacement` 2 `` ``;\
mapkey(continued) ~ Select `Odui_Dlg_00` `pg_vis_tab` 1 `tab_2`;\
mapkey(continued) ~ Select `Odui_Dlg_00` `pg_vis_tab` 1 `tab_1`;@PAUSE_FOR_SCREEN_PICK;\
mapkey(continued) ~ Select `Odui_Dlg_00` `pg_vis_tab` 1 `tab_2`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `0` `ref`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `` ``;\
mapkey(continued) ~ Open `Odui_Dlg_00` `t2.AxisMenu1`;~ Close `Odui_Dlg_00` `t2.AxisMenu1`;\
mapkey(continued) ~ Select `Odui_Dlg_00` `t2.AxisMenu1` 1 `Axis_Y`;\
mapkey(continued) ~ Activate `Odui_Dlg_00` `t2.FlipBtn1`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable2` 2 `DuMmY` `ref2`;\
mapkey(continued) ~ Focus `Odui_Dlg_00` `t2.DirectionTable2`;\
mapkey(continued) ~ FocusIn `Odui_Dlg_00` `t2.DirectionTable2`;\
mapkey(continued) ~ Select `Odui_Dlg_00` `t2.DirectionTable2` 2 `DuMmY` `ref2`;\
mapkey(continued) ~ Timer `` `` `CollectorWdg_FocusTimer`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable2` 2 `` ``;\
mapkey(continued) ~ FocusOut `Odui_Dlg_00` `t2.DirectionTable2`;@PAUSE_FOR_SCREEN_PICK;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `0` `ref`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `` ``;\
mapkey(continued) ~ Activate `Odui_Dlg_00` `t2.FlipBtn2`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `0` `ref`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable1` 2 `` ``;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable2` 2 `0` `ref2`;\
mapkey(continued) ~ Trigger `Odui_Dlg_00` `t2.DirectionTable2` 2 `` ``;\
mapkey(continued) ~ Activate `Odui_Dlg_00` `stdbtn_1`;
